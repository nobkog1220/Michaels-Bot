<script src="https://cdn.jsdelivr.net/npm/marked/marked.min.js"></script>

<script>
  (function() {
    const BACKEND_URL = "https://michaels-chatbot.up.railway.app/chat";

    const messagesDiv = document.getElementById("mb-messages");
    const userInput = document.getElementById("mb-user-input");
    const sendBtn = document.getElementById("mb-send-btn");
    const toggleBtn = document.getElementById("mb-toggle-btn");
    const closeBtn = document.getElementById("mb-close-btn");
    const chatWidget = document.getElementById("mb-chat-widget");

    let conversation = [];
    let sessionId = crypto.randomUUID();

    function addMessage(text, sender) {
      const messageDiv = document.createElement("div");
      messageDiv.className = "mb-message mb-" + sender + "-message";
      if (sender === "bot") {
        messageDiv.innerHTML = marked.parse(text);
      } else {
        messageDiv.textContent = text;
      }
      messagesDiv.appendChild(messageDiv);
      messagesDiv.scrollTop = messagesDiv.scrollHeight;
    }

    async function sendMessage() {
      const text = userInput.value.trim();
      if (!text) return;

      addMessage(text, "user");
      conversation.push({role: "user", content: text});
      userInput.value = "";
      sendBtn.disabled = true;

      try {
        const response = await fetch(BACKEND_URL, {
          method: "POST",
          headers: {"Content-Type": "application/json"},
          body: JSON.stringify({
            conversation: conversation,
            session_id: sessionId
          })
        });

        const data = await response.json();
        addMessage(data.reply, "bot");
        conversation.push({role: "assistant", content: data.reply});
      } catch (error) {
        addMessage("Sorry, having trouble connecting right now. Please try again, or call us at (847) 432-3338.", "bot");
        console.error(error);
      }

      sendBtn.disabled = false;
      userInput.focus();
    }

    sendBtn.addEventListener("click", sendMessage);
    userInput.addEventListener("keypress", function(event) {
      if (event.key === "Enter") {
        sendMessage();
      }
    });

    toggleBtn.addEventListener("click", function() {
      chatWidget.classList.toggle("mb-hidden");
      if (!chatWidget.classList.contains("mb-hidden")) {
        userInput.focus();
      }
    });

    closeBtn.addEventListener("click", function() {
      chatWidget.classList.add("mb-hidden");
    });
  })();
</script>
